cmake_minimum_required(VERSION 3.24)
project(MansurIME VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/W4 /utf-8 /permissive- /EHsc)
  add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
add_library(mansur_core STATIC core/core.cpp)
target_include_directories(mansur_core PUBLIC include)
include(CTest)
if(BUILD_TESTING)
  add_executable(mansur_core_tests tests/core_tests.cpp)
  target_link_libraries(mansur_core_tests PRIVATE mansur_core)
  add_test(NAME core COMMAND mansur_core_tests)
  # Explicit real-model smoke runner consumes only these two fixed confirmations.
  # Not a CTest: it requires configured local models and an audio device.
  add_executable(mansur_local_learning_source tests/local_learning_source_probe.cpp)
  target_link_libraries(mansur_local_learning_source PRIVATE mansur_core)
endif()
add_executable(mansur_core_probe tools/core_probe.cpp)
target_link_libraries(mansur_core_probe PRIVATE mansur_core)
add_executable(mansur_lexicon_compile tools/lexicon_compile.cpp)
target_link_libraries(mansur_lexicon_compile PRIVATE mansur_core)
option(MANSUR_BUILD_WINDOWS "Build Windows TSF integration" ON)
if(WIN32 AND MANSUR_BUILD_WINDOWS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/windows/CMakeLists.txt")
  add_subdirectory(windows)
  add_executable(mansur_register installer/register.cpp)
  target_link_libraries(mansur_register PRIVATE ole32 oleaut32 uuid advapi32)
  if(BUILD_TESTING)
    # Run explicitly with the expected installed DLL path. Activates the profile
    # only in its private test thread (flags=0); no external UI/default changes.
    add_executable(mansur_tsf_activation_host tests/tsf_activation_host.cpp)
    target_link_libraries(mansur_tsf_activation_host PRIVATE ole32 oleaut32 uuid user32)
  endif()
endif()
